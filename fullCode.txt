        private void InitializeForm()
        {
            try

        private void LoadBooks()

        private void DisplayBooks(List<BookDto> books)
        {
            var table = new DataTable();
            table.Columns.Add("ID", typeof(int));
            table.Columns.Add("Название", typeof(string));
            table.Columns.Add("Автор", typeof(string));
            table.Columns.Add("Цена", typeof(decimal));
            table.Columns.Add("Жанр", typeof(string));
            table.Columns.Add("Год", typeof(int));
            table.Columns.Add("В наличии", typeof(int));
            table.Columns.Add("Статус", typeof(string));

            foreach (var book in books)
            {
                table.Rows.Add(
                    book.Id,
                    book.Title,
                    book.AuthorName,
                    book.Price,
                    book.Genre ?? "",
                    book.PublishedYear,
                    book.StockQuantity ?? 0,
                    book.Availability
                );
            }

            dgvBooks.DataSource = table;
            FormatBooksGrid();
            lblBookCount.Text = $"Найдено книг: {books.Count}";
        }

        private void FormatBooksGrid()
        {
            if (dgvBooks.Columns.Count > 0)
            {
                dgvBooks.Columns["ID"].Visible = false;
                dgvBooks.Columns["Цена"].DefaultCellStyle.Format = "N2";

                foreach (DataGridViewRow row in dgvBooks.Rows)
                {
                    if (row.Cells["В наличии"]?.Value != null)
                    {
                        int stock = Convert.ToInt32(row.Cells["В наличии"].Value);
                        if (stock == 0)
                        {
                            row.DefaultCellStyle.ForeColor = Color.Gray;
                            row.DefaultCellStyle.BackColor = Color.LightGray;
                        }
                        else if (stock < 5)
                        {
                            row.DefaultCellStyle.ForeColor = Color.Orange;
                        }
                    }
                }
            }
        }

        private void UpdateCartDisplay()
        {
            var table = new DataTable();
            table.Columns.Add("BookId", typeof(int));
            table.Columns.Add("Title", typeof(string));
            table.Columns.Add("Author", typeof(string));
            table.Columns.Add("Price", typeof(decimal));
            table.Columns.Add("Quantity", typeof(int));
            table.Columns.Add("Subtotal", typeof(decimal));

            foreach (var item in _cart)
            {
                table.Rows.Add(
                    item.BookId,
                    item.Title,
                    item.Author,
                    item.Price,
                    item.Quantity,
                    item.Subtotal
                );
            }

            dgvCart.DataSource = table;

            if (dgvCart.Columns.Count > 0)
            {
                dgvCart.Columns["BookId"].Visible = false;
                dgvCart.Columns["Price"].DefaultCellStyle.Format = "N2";
                dgvCart.Columns["Subtotal"].DefaultCellStyle.Format = "N2";
            }

            int totalItems = _cart.Sum(item => item.Quantity);
            decimal totalAmount = _cart.Sum(item => item.Subtotal);

            lblCartItems.Text = $"Товаров в корзине: {totalItems}";
            lblCartTotal.Text = $"Общая сумма: {totalAmount:N2} руб.";

            btnPlaceOrder.Enabled = _cart.Count > 0;
        }

        private void btnAddToCart_Click(object sender, EventArgs e)
        {
            if (dgvBooks.CurrentRow == null) return;

            try
            {
                int bookId = Convert.ToInt32(dgvBooks.CurrentRow.Cells["ID"].Value);
                var book = _bookService.GetBookById(bookId);

                if (book == null)
                {
                    MessageBox.Show("Книга не найдена", "Ошибка",
                        MessageBoxButtons.OK, MessageBoxIcon.Error);
                    return;
                }

                if ((book.StockQuantity ?? 0) <= 0)
                {
                    MessageBox.Show("Эта книга временно отсутствует на складе",
                        "Внимание", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return;
                }

                // Проверяем, есть ли уже в корзине
                var existingItem = _cart.FirstOrDefault(item => item.BookId == bookId);
                if (existingItem != null)
                {
                    if (existingItem.Quantity < (book.StockQuantity ?? 0))
                    {
                        existingItem.Quantity++;
                        existingItem.Subtotal = existingItem.Price * existingItem.Quantity;
                    }
                    else
                    {
                        MessageBox.Show($"Нельзя добавить больше книг. В наличии только {book.StockQuantity} шт.",
                            "Внимание", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                        return;
                    }
                }
                else
                {
                    _cart.Add(new CartItem
                    {
                        BookId = book.Id,
                        Title = book.Title,
                        Author = book.AuthorName,
                        Price = book.Price,
                        Quantity = 1,
                        Subtotal = book.Price
                    });
                }

                UpdateCartDisplay();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка добавления в корзину: {ex.Message}", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnRemoveFromCart_Click(object sender, EventArgs e)
        {
            if (dgvCart.CurrentRow == null) return;

            try
            {
                int bookId = Convert.ToInt32(dgvCart.CurrentRow.Cells["BookId"].Value);
                var cartItem = _cart.FirstOrDefault(item => item.BookId == bookId);

                if (cartItem != null)
                {
                    if (cartItem.Quantity > 1)
                    {
                        cartItem.Quantity--;
                        cartItem.Subtotal = cartItem.Price * cartItem.Quantity;
                    }
                    else
                    {
                        _cart.Remove(cartItem);
                    }

                    UpdateCartDisplay();
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка удаления из корзины: {ex.Message}", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnClearCart_Click(object sender, EventArgs e)
        {
            if (_cart.Count > 0)
            {
                var result = MessageBox.Show("Очистить корзину?", "Подтверждение",
                    MessageBoxButtons.YesNo, MessageBoxIcon.Question);

                if (result == DialogResult.Yes)
                {
                    _cart.Clear();
                    UpdateCartDisplay();
                }
            }
        }

        private void btnPlaceOrder_Click(object sender, EventArgs e)
        {
            if (_currentUser == null)
            {
                MessageBox.Show("Пользователь не выбран", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            if (_cart.Count == 0)
            {
                MessageBox.Show("Корзина пуста", "Внимание",
                    MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            try
            {
                var orderDto = new CreateOrderDto
                {
                    UserId = _currentUser.Id,
                    Items = _cart.Select(item => new OrderItemCreateDto
                    {
                        BookId = item.BookId,
                        Quantity = item.Quantity,
                        Price = item.Price
                    }).ToList()
                };

                var result = _orderService.PlaceOrder(orderDto);

                if (result.Success)
                {
                    MessageBox.Show(result.Message, "Заказ оформлен",
                        MessageBoxButtons.OK, MessageBoxIcon.Information);

                    _cart.Clear();
                    UpdateCartDisplay();
                    LoadBooks(); // Обновляем список книг
                }
                else
                {
                    MessageBox.Show(result.Message, "Ошибка",
                        MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка оформления заказа: {ex.Message}", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void SearchBooks()
        {
            try
            {
                string title = txtBookSearch.Text.Trim();
                string genre = cmbGenreFilter.SelectedItem?.ToString();

                if (string.IsNullOrEmpty(title) && (genre == "Все жанры" || string.IsNullOrEmpty(genre)))
                {
                    LoadBooks();
                }
                else
                {
                    var books = _bookService.SearchBooks(
                        string.IsNullOrEmpty(title) ? null : title,
                        null,
                        genre == "Все жанры" ? null : genre
                    );
                    DisplayBooks(books);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка поиска: {ex.Message}", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void SearchByAuthor()
        {
            try
            {
                string authorName = txtAuthorSearch.Text.Trim();
                decimal minPrice = numMinPrice.Value;

                if (string.IsNullOrEmpty(authorName))
                {
                    MessageBox.Show("Введите имя автора", "Внимание",
                        MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return;

                    title: null,
                    author: authorName,
                    genre: null
                );

                // Фильтруем по минимальной цене

                table.Columns.Add("В наличии", typeof(int));

                foreach (var book in books)
                {
                    table.Rows.Add(
                        book.Id,
                        book.Title,
                        book.AuthorName,
                        book.Price,
                        book.Genre ?? "",
                        book.PublishedYear,
                        book.StockQuantity ?? 0

                dgvAuthorSearch.DataSource = table;

                if (dgvAuthorSearch.Columns.Count > 0)

                    dgvAuthorSearch.Columns["ID"].Visible = false;
                    dgvAuthorSearch.Columns["Цена"].DefaultCellStyle.Format = "N2";

                MessageBox.Show($"Ошибка поиска по автору: {ex.Message}", "Ошибка",

        }

        private void LoadUserOrders()
        {
            if (_currentUser == null) return;

            try
            {
                var orders = _orderService.GetUserOrders(_currentUser.Id);

                var table = new DataTable();
                table.Columns.Add("ID", typeof(int));
                table.Columns.Add("Дата", typeof(string));
                table.Columns.Add("Время", typeof(string));
                table.Columns.Add("Сумма", typeof(decimal));
                table.Columns.Add("Статус", typeof(string));
                table.Columns.Add("Товаров", typeof(int));

                foreach (var order in orders)
                {
                    string dateStr = order.OrderDate.ToString("dd.MM.yyyy");
                    string timeStr = order.OrderTime.ToString("HH:mm");

                    table.Rows.Add(
                        order.Id,
                        dateStr,
                        timeStr,
                        order.TotalAmount,
                        order.Status,
                        order.Items.Count
                    );
                }

                dgvOrders.DataSource = table;

                if (dgvOrders.Columns.Count > 0)
                {
                    dgvOrders.Columns["Сумма"].DefaultCellStyle.Format = "N2";

                    // Раскрашиваем статусы
                    foreach (DataGridViewRow row in dgvOrders.Rows)
                    {
                        if (row.Cells["Статус"]?.Value != null)
                        {
                            string status = row.Cells["Статус"].Value.ToString();
                            if (status == "Выполнен")
                                row.DefaultCellStyle.ForeColor = Color.Green;
                            else if (status == "Отменен")
                                row.DefaultCellStyle.ForeColor = Color.Red;
                            else if (status == "В обработке")
                                row.DefaultCellStyle.ForeColor = Color.Blue;
                        }
                    }
                }

                lblOrderCount.Text = $"Заказов: {orders.Count}";
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка загрузки заказов: {ex.Message}", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnLoadOrders_Click(object sender, EventArgs e)

        private void EnableEditMode(bool enable, bool isNew = false)
        {
            groupBoxBookDetails.Enabled = enable;
            btnSaveBook.Enabled = enable;
            btnCancelEdit.Enabled = enable;

            if (enable)
            {
                btnAddNewBook.Enabled = false;
                btnEditBook.Enabled = false;
                btnDeleteBook.Enabled = false;

                if (isNew)
                {
                    ClearBookForm();
                    _isEditMode = false;
                    _editingBookId = 0;
                    lblManageStatus.Text = "Режим добавления новой книги";
                    lblManageStatus.ForeColor = Color.Blue;
                }
                else
                {
                    _isEditMode = true;
                    lblManageStatus.Text = "Режим редактирования книги";
                    lblManageStatus.ForeColor = Color.Orange;
                }
            }
            else
            {
                btnAddNewBook.Enabled = true;
                btnEditBook.Enabled = true;
                btnDeleteBook.Enabled = true;
                _isEditMode = false;
                _editingBookId = 0;
                ClearBookForm();
                lblManageStatus.Text = "Готов к работе";
                lblManageStatus.ForeColor = Color.Green;
            }
        }

        private void btnEditBook_Click(object sender, EventArgs e)

                var book = _bookService.GetBookById(bookId);

                if (book == null)

                _editingBookId = book.Id;
                txtBookTitle.Text = book.Title;
                numBookPrice.Value = book.Price;
                cmbBookAuthor.Text = book.AuthorName;
                numPublishedYear.Value = book.PublishedYear;
                txtBookGenre.Text = book.Genre ?? "";
                numStockQuantity.Value = book.StockQuantity ?? 1;

                EnableEditMode(true, false);
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка загрузки данных книги: {ex.Message}", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnDeleteBook_Click(object sender, EventArgs e)
        {
            if (dgvManageBooks.CurrentRow == null)
            {
                MessageBox.Show("Выберите книгу для удаления", "Внимание",
                    MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            var result = MessageBox.Show("Вы уверены, что хотите удалить выбранную книгу?",
                "Подтверждение удаления", MessageBoxButtons.YesNo, MessageBoxIcon.Question);

            if (result == DialogResult.Yes)
            {
                try
                {
                    int bookId = Convert.ToInt32(dgvManageBooks.CurrentRow.Cells["ID"].Value);
                    bool success = _bookService.DeleteBook(bookId);

                    if (success)
                    {
                        MessageBox.Show("Книга успешно удалена!", "Успех",
                            MessageBoxButtons.OK, MessageBoxIcon.Information);
                        LoadManageBooks();
                        LoadBooks(); // Обновляем основной каталог
                    }
                    else
                    {
                        MessageBox.Show("Не удалось удалить книгу", "Ошибка",
                            MessageBoxButtons.OK, MessageBoxIcon.Error);
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Ошибка удаления книги: {ex.Message}", "Ошибка",
                        MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
        }

        private void btnSaveBook_Click(object sender, EventArgs e)
        {
            SaveBook();
        }

        private void SaveBook()
        {
            if (string.IsNullOrWhiteSpace(txtBookTitle.Text))
            {
                MessageBox.Show("Введите название книги", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Warning);
                txtBookTitle.Focus();
                return;
            }

            if (string.IsNullOrWhiteSpace(cmbBookAuthor.Text))
            {
                MessageBox.Show("Выберите автора", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Warning);
                cmbBookAuthor.Focus();
                return;
            }

            try
            {
                if (_isEditMode && _editingBookId > 0)
                {
                    var bookDto = new BookUpdateDto
                    {
                        Id = _editingBookId,
                        Title = txtBookTitle.Text,
                        AuthorName = cmbBookAuthor.Text,
                        Price = numBookPrice.Value,
                        Genre = txtBookGenre.Text,
                        PublishedYear = (int)numPublishedYear.Value,
                        StockQuantity = (int)numStockQuantity.Value
                    };

                    var updatedBook = _bookService.UpdateBook(bookDto);
                    MessageBox.Show($"Книга '{updatedBook.Title}' успешно обновлена!", "Успех",
                        MessageBoxButtons.OK, MessageBoxIcon.Information);
                }
                else
                {
                    var bookDto = new BookCreateDto
                    {
                        Title = txtBookTitle.Text,
                        AuthorName = cmbBookAuthor.Text,
                        Price = numBookPrice.Value,
                        Genre = txtBookGenre.Text,
                        PublishedYear = (int)numPublishedYear.Value,
                        StockQuantity = (int)numStockQuantity.Value
                    };

                    var createdBook = _bookService.AddBook(bookDto);
                    MessageBox.Show($"Книга '{createdBook.Title}' успешно добавлена! ID: {createdBook.Id}", "Успех",
                        MessageBoxButtons.OK, MessageBoxIcon.Information);
                }

                EnableEditMode(false);
                LoadManageBooks();
                LoadBooks(); // Обновляем основной каталог
                InitializeGenreFilter(); // Обновляем фильтры
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка сохранения книги: {ex.Message}", "Ошибка",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void dgvManageBooks_SelectionChanged(object sender, EventArgs e)
        {
            if (_isEditMode && dgvManageBooks.CurrentRow != null)
            {
                var result = MessageBox.Show("Сохранить текущие изменения?", "Внимание",
                    MessageBoxButtons.YesNoCancel, MessageBoxIcon.Question);

                if (result == DialogResult.Yes)
                {
                    SaveBook();
                }
                else if (result == DialogResult.No)
                {
                    EnableEditMode(false);
                }
                // Если Cancel - ничего не делаем
            }
        }

                    EnableEditMode(false);

            }
        }

using BookStore.BLL.DTO.Order;
using BookStore.DAL.Data;
using BookStore.DAL.Models;
using Microsoft.EntityFrameworkCore;
using System;
using System.Collections.Generic;
using System.Linq;

namespace BookStore.BLL.Services
{
    public class OrderService
    {
        private readonly string _connectionString;

        public OrderService(string connectionString)
        {
            _connectionString = connectionString;
        }

        public OrderResultDto PlaceOrder(CreateOrderDto orderDto)
        {
            var result = new OrderResultDto();

            using var context = new BookstoreContext();
            using var transaction = context.Database.BeginTransaction();

            try
            {
                // 1. Проверяем пользователя
                var user = context.Users.Find(orderDto.UserId);
                if (user == null)
                {
                    result.Success = false;
                    result.Message = "Пользователь не найден";
                    return result;
                }

                // 2. Проверяем наличие товаров
                foreach (var item in orderDto.Items)
                {
                    var book = context.Books.Find(item.BookId);
                    if (book == null || (book.StockQuantity ?? 0) < item.Quantity)
                    {
                        result.Success = false;
                        result.Message = $"Недостаточно товара: {book?.Title ?? "Неизвестная книга"}";
                        return result;
                    }
                }

                // 3. Создаем заказ - используем DateOnly и TimeOnly
                var order = new Order
                {
                    UserId = orderDto.UserId,
                    OrderDate = DateOnly.FromDateTime(DateTime.Now),  // Исправлено
                    OrderTime = TimeOnly.FromDateTime(DateTime.Now),  // Исправлено
                    TotalAmount = orderDto.Items.Sum(i => i.Price * i.Quantity),
                    Status = "Новый",
                    CreatedAt = DateTime.SpecifyKind(DateTime.UtcNow, DateTimeKind.Unspecified)
                };

                context.Orders.Add(order);
                context.SaveChanges(); // Получаем Id заказа

                // 4. Добавляем позиции заказа и обновляем остатки
                foreach (var item in orderDto.Items)
                {
                    var orderItem = new OrderItem
                    {
                        OrderId = order.Id,
                        BookId = item.BookId,
                        Quantity = item.Quantity,
                        UnitPrice = item.Price,
                        Subtotal = item.Price * item.Quantity
                    };

                    context.OrderItems.Add(orderItem);

                    // Обновляем остаток на складе
                    var book = context.Books.Find(item.BookId);
                    if (book != null)
                    {
                        book.StockQuantity = (book.StockQuantity ?? 0) - item.Quantity;
                        if (book.StockQuantity < 0) book.StockQuantity = 0;
                    }
                }

                context.SaveChanges();
                transaction.Commit();

                result.Success = true;
                result.OrderId = order.Id;
                result.OrderNumber = $"ORD-{order.Id:000000}";
                result.TotalAmount = order.TotalAmount;
                result.Message = $"Заказ №{order.Id} успешно оформлен!";
            }
            catch (Exception ex)
            {
                transaction.Rollback();
                result.Success = false;
                result.Message = $"Ошибка при оформлении заказа: {ex.Message} Inner: {ex.InnerException?.Message}";
            }

                return result;
        }

        public List<OrderDto> GetUserOrders(int userId)
        {
            using var context = new BookstoreContext();

            var orders = context.Orders
                .Include(o => o.User)
                .Include(o => o.OrderItems)
                    .ThenInclude(oi => oi.Book)
                        .ThenInclude(b => b.Author)
                .Where(o => o.UserId == userId)
                .OrderByDescending(o => o.OrderDate)
                .ThenByDescending(o => o.OrderTime)
                .ToList();

            return orders.Select(o => new OrderDto
            {
                Id = o.Id,
                UserId = o.UserId,
                UserName = o.User?.FullName ?? "Неизвестный",
                OrderDate = o.OrderDate,
                OrderTime = o.OrderTime,
                TotalAmount = o.TotalAmount,
                Status = o.Status ?? "Новый",
                CreatedAt = o.CreatedAt,
                Items = o.OrderItems.Select(oi => new OrderItemDto
                {
                    BookId = oi.BookId ?? 0,
                    BookTitle = oi.Book?.Title ?? "Неизвестная книга",
                    AuthorName = oi.Book?.Author?.Name ?? "Неизвестный автор",
                    Quantity = oi.Quantity,
                    UnitPrice = oi.UnitPrice,
                    Subtotal = oi.Subtotal
                }).ToList()
            }).ToList();
        }

                UserId = order.UserId,
                UserName = order.User?.FullName ?? "Неизвестный",
                OrderDate = order.OrderDate,
                OrderTime = order.OrderTime,
                TotalAmount = order.TotalAmount,
                Status = order.Status ?? "Новый",
                CreatedAt = order.CreatedAt,
                Items = order.OrderItems.Select(oi => new OrderItemDto
                {
                    BookId = oi.BookId ?? 0,
                    BookTitle = oi.Book?.Title ?? "Неизвестная книга",
                    AuthorName = oi.Book?.Author?.Name ?? "Неизвестный автор",
                    Quantity = oi.Quantity,
                    UnitPrice = oi.UnitPrice,
                    Subtotal = oi.Subtotal
                }).ToList()
            };
        }

        public bool CancelOrder(int orderId)
        {
            using var context = new BookstoreContext();
            using var transaction = context.Database.BeginTransaction();

            try
            {
                var order = context.Orders
                    .Include(o => o.OrderItems)
                    .FirstOrDefault(o => o.Id == orderId);

                if (order == null) return false;

                // Возвращаем товары на склад
                foreach (var item in order.OrderItems)
                {
                    var book = context.Books.Find(item.BookId);
                    if (book != null)
                    {
                        book.StockQuantity = (book.StockQuantity ?? 0) + item.Quantity;
                    }
                }

                order.Status = "Отменен";
                context.SaveChanges();
                transaction.Commit();

                return true;
            }
            catch
            {
                transaction.Rollback();
                return false;
            }
        }

                .Where(b => !string.IsNullOrEmpty(b.Genre))
                .Select(b => b.Genre!)
                .Distinct()
                .OrderBy(g => g)
                .ToList();
        }

        {
            using var context = new BookstoreContext();

            return context.Authors
                .Select(a => a.Name)
                .Distinct()
                .OrderBy(a => a)
                .ToList();
        }

        public List<BookDto> SearchBooks(string? title = null, string? author = null, string? genre = null)
        {
            using var context = new BookstoreContext();

            var query = context.Books
                .Include(b => b.Author)
                .AsQueryable();

            if (!string.IsNullOrEmpty(title))
                query = query.Where(b => b.Title.Contains(title));

            if (!string.IsNullOrEmpty(author))
                query = query.Where(b => b.Author != null && b.Author.Name.Contains(author));

            if (!string.IsNullOrEmpty(genre))
                query = query.Where(b => b.Genre != null && b.Genre.Contains(genre));

            return query
                .OrderBy(b => b.Title)
                .Select(book => new BookDto
                {
                    Id = book.Id,
                    Title = book.Title,
                    AuthorName = book.Author != null ? book.Author.Name : "Неизвестный",
                    AuthorId = book.AuthorId,
                    Price = book.Price,
                    Genre = book.Genre,
                    PublishedYear = book.Published.HasValue ? book.Published.Value.Year : 0,  // Исправлено
                    StockQuantity = book.StockQuantity,
                    Availability = (book.StockQuantity ?? 0) > 0 ? "В наличии" : "Нет в наличии",
                    CreatedAt = book.CreatedAt
                })
                .ToList();
        }

            return true;

        public BookDto AddBook(BookCreateDto bookDto)
        {

            // Находим или создаем автора
            var author = context.Authors
                .FirstOrDefault(a => a.Name == bookDto.AuthorName);

            if (author == null)
            {
                author = new Author { Name = bookDto.AuthorName };
                context.Authors.Add(author);
                context.SaveChanges();
            }

            // Создаем книгу - используем DateOnly для поля Published
            var book = new Book
            {
                Title = bookDto.Title,
                AuthorId = author.Id,
                Price = bookDto.Price,
                Genre = bookDto.Genre,
